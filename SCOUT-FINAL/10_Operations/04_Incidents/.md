<%*
/**
 * Incident Bootstrap Template
 * - Creates incident folder structure
 * - Creates INC / INV / TL from templates
 * - Opens main incident file
 * - Expands new incident folder in file explorer
 * - Deletes this scratch note
 */

// 1) Prompt for incident number
const year = tp.date.now("YYYY");
const num = await tp.system.prompt("Incident Number (digits only, e.g. 0003)?");

if (!num) {
  new Notice("Incident Bootstrap cancelled: no number provided.");
  return;
}

const padded = num.padStart(4, "0");
const incidentId = `INC-${year}-${padded}`;

// 2) Define base paths for this incident
const incidentsRoot = "10_Operations/04_Incidents";   // match your vault
const base = `${incidentsRoot}/${incidentId}`;

// helper: ensure folder exists
async function ensureFolder(path) {
  if (!app.vault.getAbstractFileByPath(path)) {
    await app.vault.createFolder(path);
  }
}

// 3) Create the incident folder and supporting subfolders
await ensureFolder(incidentsRoot);
await ensureFolder(base);
await ensureFolder(`${base}/Artifacts`);
await ensureFolder(`${base}/Evidence`);
await ensureFolder(`${base}/Response_Actions`);
await ensureFolder(`${base}/Observations`);
await ensureFolder(`${base}/Investigation`);
await ensureFolder(`${base}/Timeline`);

// optional: create .keep markers (now that folders exist)
await app.vault.create(`${base}/Artifacts/.keep`, "");
await app.vault.create(`${base}/Evidence/.keep`, "");
await app.vault.create(`${base}/Response_Actions/.keep`, "");
await app.vault.create(`${base}/Observations/.keep`, "");

// 4) Create the INCIDENT note from your Incident template
const incidentTemplate = tp.file.find_tfile("00_System/00_Templates/02_Operations/Incident.md Template");
// ^^^ adjust if the path/name differs

// create_new(template, filename, open_new, folder)
const incidentFile = await tp.file.create_new(
  incidentTemplate,
  incidentId,
  false,
  base
);

// 5) Create Investigation note from your Investigation template
const invId = `INV-${year}-${padded}`;
const invTemplate = tp.file.find_tfile("00_System/00_Templates/02_Operations/Investigation.md Template");
// ^^^ adjust to your actual Investigation template path/name

await tp.file.create_new(
  invTemplate,
  invId,
  false,
  `${base}/Investigation`
);

// 6) Create Timeline note from your Timeline template
const tlId = `TL-${year}-${padded}`;
const tlTemplate = tp.file.find_tfile("00_System/00_Templates/02_Operations/Timeline.md Template");
// ^^^ adjust to your actual Timeline template path/name

await tp.file.create_new(
  tlTemplate,
  tlId,
  false,
  `${base}/Timeline`
);

// 7) Append handy links into the INCIDENT note
const links = `

# Incident: ${incidentId}

- Investigation: [[${invId}]]
- Timeline: [[${tlId}]]

`;

let currentContent = await app.vault.read(incidentFile);
await app.vault.modify(incidentFile, currentContent + links);

// 8) Open the main INCIDENT file in the active pane
const leaf = app.workspace.getLeaf(false);     // current leaf
await leaf.openFile(incidentFile);

// 9) Reveal the INCIDENT file/folder in the file explorer
const explorerLeaves = app.workspace.getLeavesOfType("file-explorer");
if (explorerLeaves && explorerLeaves.length > 0) {
  const explorer = explorerLeaves[0].view;
  if (explorer && explorer.revealInFolder) {
    explorer.revealInFolder(incidentFile);
  }
}

// 10) Delete this scratch note (the bootstrap note created by "Create new note from template")
const bootstrapNote = tp.file.find_tfile(tp.file.path);
await app.vault.delete(bootstrapNote);

%>
