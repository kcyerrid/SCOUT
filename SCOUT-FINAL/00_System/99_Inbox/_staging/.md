<%*
/**
 * ITID Bootstrap (hardened: creates ITID note from template + 5s janitor for 40_ITIDs/Untitled*.md)
 *
 * Template used:
 * 00_System/00_Templates/04_ITID/ITID_template.md
 */

const ITID_ROOT = "40_ITIDs/01_Definitions";
const ITID_TEMPLATE_PATH = "00_System/00_Templates/04_ITID/ITID_template.md";

const YAML_KEY_ID = "ITID_ID";
const YAML_KEY_TITLE = "ITID_Title";

// ---------------- helpers ----------------
function norm(s){ return String(s||"").replace(/\\/g,"/"); }
function isFolder(af){ return !!af && Array.isArray(af.children); }
function isFile(af){ return !!af && !Array.isArray(af.children); }

function sanitizeFilename(name){
  return String(name||"")
    .replace(/[\\/:*?"<>|]/g, "-")
    .replace(/\s+/g, " ")
    .trim();
}

function stripMd(name){
  return String(name || "").replace(/\.md$/i, "").trim();
}

function normalizeText(s){
  s = (s || "").replace(/\r\n/g, "\n");
  if (s && s.length && s.charCodeAt(0) === 0xFEFF) s = s.slice(1);
  return s;
}

function extractFrontmatterFlexible(text){
  const t = normalizeText(text);
  const m = t.match(/^(\s*)---\n([\s\S]*?)\n---\n?/);
  if (!m) return null;
  return { prefix: m[1] || "", body: m[2], after: t.slice(m[0].length) };
}

function setYamlKey(fmBody, key, value){
  const re = new RegExp("^" + key + ":[^\\n]*$", "m");
  const line = key + ": " + value;
  if (re.test(fmBody)) return fmBody.replace(re, line);
  return fmBody + "\n" + line;
}

function buildDoc(prefix, fmBody, after){
  return (prefix || "") + "---\n" + fmBody + "\n---\n" + (after || "");
}

function replaceFirstH1(text, newH1){
  let t = normalizeText(text);
  const re = /^# .*$\n?/m;
  if (re.test(t)) return t.replace(re, "# " + newH1 + "\n");

  const m = t.match(/^\s*---\n[\s\S]*?\n---\n?/);
  if (m) {
    const idx = t.indexOf(m[0]) + m[0].length;
    return t.slice(0, idx) + "\n# " + newH1 + "\n\n" + t.slice(idx);
  }
  return "# " + newH1 + "\n\n" + t;
}

function parseItid(itidRaw){
  const s = String(itidRaw || "").trim().toUpperCase();
  const m = s.match(/^ITID-(\d{5})$/);
  if (!m) return null;
  return { full: s, digits: m[1] };
}

// 11113 -> [ITID-10000, ITID-11000, ITID-11100, ITID-11110]
function computeBucketChain(digits){
  const width = digits.length;
  const out = [];
  for (let k = 1; k <= width - 1; k++){
    const bucketDigits = digits.slice(0, k) + "0".repeat(width - k);
    out.push("ITID-" + bucketDigits);
  }
  return out;
}

// Extract ITID-##### from folder name by digits regardless of dash type
function folderBucketKey(name){
  const s = String(name || "").trim();
  const m = s.match(/^ITID\D*(\d{5})/i);
  if (!m) return null;
  return "ITID-" + m[1];
}

function findChildFolderByBucket(parentFolderPath, bucketId){
  const parent = app.vault.getAbstractFileByPath(norm(parentFolderPath));
  if (!isFolder(parent)) return { folder: null, ambiguous: false };

  const bid = bucketId.toUpperCase();
  let found = null;

  for (const ch of (parent.children || [])){
    if (!isFolder(ch)) continue;
    const key = folderBucketKey(ch.name);
    if (key && key.toUpperCase() === bid){
      if (found) return { folder: null, ambiguous: true };
      found = ch;
    }
  }

  return { folder: found, ambiguous: false };
}

async function focusFile(file){
  const leaf = app.workspace.getLeaf(true);
  await leaf.openFile(file, { active: true });
  app.workspace.setActiveLeaf(leaf, { focus: true });
}

function detachLeavesShowingPath(path){
  const leaves = app.workspace.getLeavesOfType("markdown") || [];
  for (const leaf of leaves){
    const f = leaf?.view?.file;
    if (f && f.path === path){
      try { leaf.detach(); } catch (e) {}
    }
  }
}

async function kill40ItidsUntitled(createdFile){
  const createdPath = createdFile?.path || "";
  const folderPath = "40_ITIDs";
  const re = /^Untitled(\s*\d*)?\.md$/i;

  const folder = app.vault.getAbstractFileByPath(folderPath);
  if (!isFolder(folder)) return;

  for (const ch of (folder.children || [])){
    if (!isFile(ch)) continue;
    if (ch.path === createdPath) continue;

    if (re.test(ch.name || "")){
      detachLeavesShowingPath(ch.path);
      try { await app.vault.delete(ch); } catch (e) {}
    }
  }
}

/**
 * Schedule repeated cleanup passes after template execution returns control to Obsidian.
 * This catches late-created Untitled.md that appears after your script completes.
 */
function scheduleJanitor(createdFile, passes = 12, intervalMs = 450){
  for (let i = 0; i < passes; i++){
    window.setTimeout(() => {
      (async () => {
        try {
          await kill40ItidsUntitled(createdFile);
          await focusFile(createdFile);
        } catch (e) {}
      })();
    }, i * intervalMs);
  }
}

// -------------------- MAIN --------------------

// Validate ITID_ROOT exists and is a folder
const root = app.vault.getAbstractFileByPath(norm(ITID_ROOT));
if (!isFolder(root)){
  new Notice(`ITID_ROOT is not a folder in Obsidian: ${ITID_ROOT}`);
  return;
}

// Validate ITID template exists
const itidTemplate = app.vault.getAbstractFileByPath(norm(ITID_TEMPLATE_PATH));
if (!itidTemplate){
  new Notice(`ITID template not found: ${ITID_TEMPLATE_PATH}`);
  return;
}

// 1) Prompt
const itidRaw = await tp.system.prompt("ITID ID (e.g., ITID-11113)");
if (!itidRaw) { new Notice("ITID cancelled."); return; }

const parsed = parseItid(itidRaw);
if (!parsed){
  new Notice("Invalid ITID format. Expected ITID-##### (5 digits).");
  return;
}

const titleRaw = await tp.system.prompt("ITID Title");
const itidTitle = (titleRaw || "").trim();

// 2) Resolve taxonomy folder (strict nested walking)
const chain = computeBucketChain(parsed.digits);
let currentPath = ITID_ROOT;
let targetFolderPath = ITID_ROOT;

for (const bucketId of chain){
  const res = findChildFolderByBucket(currentPath, bucketId);

  if (res.ambiguous){
    new Notice(`Ambiguous taxonomy: multiple folders match ${bucketId} under ${currentPath}`);
    return;
  }
  if (!res.folder){
    new Notice(`Missing category folder ${bucketId} under ${currentPath}`);
    return;
  }

  currentPath = res.folder.path;
  targetFolderPath = res.folder.path;
}

// 3) Create the real ITID note from the ITID content template
const safeTitle = sanitizeFilename(itidTitle);
const baseName = stripMd(itidTitle ? `${parsed.full} - ${safeTitle}` : parsed.full);

const createdFile = await tp.file.create_new(
  itidTemplate,
  baseName,
  false,           // don't rely on Templater's open behavior
  targetFolderPath
);

if (!createdFile){
  new Notice("Failed to create ITID note from template.");
  return;
}

// 4) Patch YAML + H1 in the created file
let text = await app.vault.read(createdFile);

let fm = extractFrontmatterFlexible(text);
if (!fm){
  fm = { prefix:"", body:"entity_type: itid", after:"\n\n" + normalizeText(text).replace(/^\s*/, "") };
}

let body = fm.body;
body = setYamlKey(body, YAML_KEY_ID, parsed.full);
body = setYamlKey(body, YAML_KEY_TITLE, `"${itidTitle.replace(/"/g,'\\"')}"`);
body = setYamlKey(body, "updated", tp.date.now("YYYY-MM-DD"));

text = buildDoc(fm.prefix, body, fm.after);
text = replaceFirstH1(text, itidTitle ? `${parsed.full} — ${itidTitle}` : `${parsed.full} — ITID`);

await app.vault.modify(createdFile, text);

// 5) Immediate cleanup + focus, then schedule janitor passes for 5–6 seconds
await focusFile(createdFile);
await kill40ItidsUntitled(createdFile);
scheduleJanitor(createdFile, 12, 450); // ~5.4 seconds total

new Notice(`ITID created: ${baseName}`);
%>
